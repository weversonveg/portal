3. JAVASCRIPT — FUNCIONAMENTO DO CAMPO
============================================================

(function () {
    function init() {
        if (typeof flatpickr === 'undefined') {
            setTimeout(init, 100);
            return;
        }

        var plugin = (typeof monthSelectPlugin === 'function') ? [new monthSelectPlugin({
            shorthand: false,
            dateFormat: "Y-m",
            altFormat: "F Y"
        })] : [];

        document.querySelectorAll('.campo-mes').forEach(function (input) {
            if (input._flatpickr) return;

            flatpickr(input, {
                locale: "pt",
                dateFormat: "Y-m",
                altInput: true,
                altFormat: "F Y",
                allowInput: false,
                clickOpens: true,
                disableMobile: true,
                plugins: plugin,
                onChange: function (selectedDates, dateStr, instance) {
                    input.dataset.value = dateStr;
                    input.dispatchEvent(new Event('change', { bubbles: true }));
                }
            });
        });

        window.setValorMes = function (input, valor) {
            if (!input) return;
            input.value = valor;
            input.dataset.value = valor;
            if (input._flatpickr) {
                try {
                    if (valor) input._flatpickr.setDate(valor, false);
                    else input._flatpickr.clear(false);
                } catch (e) {}
            }
        };

        window.lerMes = function (input) {
            if (!input) return '';
            return input.dataset.value || input.value || '';
        };

        var campoMesCalculo = document.getElementById('mesReferencia');
        if (campoMesCalculo && !campoMesCalculo.dataset.value && !campoMesCalculo.value) {
            var hoje = new Date();
            var mesAtual = hoje.getFullYear() + '-' + String(hoje.getMonth() + 1).padStart(2, '0');
            window.setValorMes(campoMesCalculo, mesAtual);
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }
})();
